# dr_stein: the Qt Quick application. One QML module ("DrStein") holds the C++
# facades from ../qt and every QML file; the Qt-free logic is drstein::core.
# 6.7 for Rectangle's per-corner radii (the linked segment controls).
find_package(Qt6 6.7 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Svg)
qt_standard_project_setup(REQUIRES 6.7)
if(QT_KNOWN_POLICY_QTP0004)
  qt_policy(SET QTP0004 NEW)   # qmldir per QML subdirectory, generated by Qt
endif()

set(DRSTEIN_QT_SOURCES
  ../qt/util.hpp
  ../qt/settings.hpp ../qt/settings.cpp
  ../qt/report_model.hpp ../qt/report_model.cpp
  ../qt/job_runner.hpp ../qt/job_runner.cpp
  ../qt/models.hpp ../qt/models.cpp
  ../qt/workspace.hpp ../qt/workspace.cpp
  ../qt/struct_inspector.hpp ../qt/struct_inspector.cpp
  ../qt/file_browser.hpp ../qt/file_browser.cpp
  ../qt/mounts_model.hpp ../qt/mounts_model.cpp
  ../qt/image_controller.hpp ../qt/image_controller.cpp
  ../qt/partition_editor.hpp ../qt/partition_editor.cpp
  ../qt/tools_controller.hpp ../qt/tools_controller.cpp
  ../qt/profiles_model.hpp ../qt/profiles_model.cpp
)

set_source_files_properties(qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)

set(DRSTEIN_PLATFORM_SOURCES
  platform/window_chrome.hpp platform/window_chrome.cpp
  platform/device_watcher.hpp platform/device_watcher.cpp
  platform/codicon_provider.hpp platform/codicon_provider.cpp
  platform/app_icon.hpp platform/app_icon.cpp)
if(APPLE)
  list(APPEND DRSTEIN_PLATFORM_SOURCES platform/window_chrome_mac.mm)
endif()

qt_add_executable(dr_stein MACOSX_BUNDLE main.cpp ${DRSTEIN_QT_SOURCES} ${DRSTEIN_PLATFORM_SOURCES})

# VS Code's caption-button glyphs for Linux (icons/codicons/README.md), served by CodiconProvider.
qt_add_resources(dr_stein codicons PREFIX "/codicons" BASE icons/codicons FILES
  icons/codicons/chrome-minimize.svg icons/codicons/chrome-maximize.svg
  icons/codicons/chrome-restore.svg icons/codicons/chrome-close.svg)

# The application icon (icons/app/README.md): the SVGs travel as resources for the window
# icon, the .icns is the bundle icon on macOS, the .ico comes in through the .rc on Windows,
# and Linux installs the SVGs into the hicolor theme next to a .desktop file.
qt_add_resources(dr_stein appicon PREFIX "/icons" BASE icons/app FILES
  icons/app/dr-stein.svg icons/app/dr-stein-small.svg)
if(APPLE)
  target_sources(dr_stein PRIVATE icons/app/dr-stein.icns)
  set_source_files_properties(icons/app/dr-stein.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
elseif(WIN32)
  enable_language(RC)
  target_sources(dr_stein PRIVATE platform/dr_stein.rc)
elseif(UNIX)
  include(GNUInstallDirs)
  install(FILES icons/app/dr-stein.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
  install(FILES icons/app/dr-stein-symbolic.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/symbolic/apps)
  install(FILES platform/dr-stein.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
endif()

qt_add_qml_module(dr_stein
  URI DrStein
  VERSION 1.0
  QML_FILES ${DRSTEIN_QML_FILES}
)

target_include_directories(dr_stein PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../qt ${CMAKE_CURRENT_SOURCE_DIR}/platform)
if(APPLE)
  target_link_libraries(dr_stein PRIVATE "-framework AppKit")
endif()
# libstein has a slots() method; Qt's keyword macros would eat it.
target_compile_definitions(dr_stein PRIVATE QT_NO_KEYWORDS)
target_link_libraries(dr_stein PRIVATE
  drstein::core drstein_warnings
  Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Svg)
set_target_properties(dr_stein PROPERTIES
  OUTPUT_NAME "Dr Stein"
  MACOSX_BUNDLE_BUNDLE_NAME "Dr Stein"
  MACOSX_BUNDLE_GUI_IDENTIFIER "local.dr-stein.app"
  MACOSX_BUNDLE_ICON_FILE dr-stein.icns
  MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
  MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION}
  WIN32_EXECUTABLE TRUE)

install(TARGETS dr_stein BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
